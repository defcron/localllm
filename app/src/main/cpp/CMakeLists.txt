cmake_minimum_required(VERSION 3.29)
project(llama_jni C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt" OR
   NOT EXISTS "${VENDOR}/opencl_dispatch.cpp" OR NOT EXISTS "${VENDOR}/vulkan-include/vulkan/vulkan.hpp")
    message(FATAL_ERROR "Native dependencies missing: run ./setup.sh in LocalLlm (CI does this automatically).")
endif()
# No hard ELF dependency on a proprietary OpenCL driver. Vulkan loads independently.
add_library(OpenCL STATIC "${VENDOR}/opencl_dispatch.cpp")
target_include_directories(OpenCL PUBLIC "${VENDOR}/opencl-include")
target_link_libraries(OpenCL PUBLIC dl)
add_library(OpenCL::OpenCL ALIAS OpenCL)
set(OpenCL_INCLUDE_DIR "${VENDOR}/opencl-include" CACHE PATH "" FORCE)
set(OpenCL_LIBRARY OpenCL CACHE STRING "" FORCE)
find_library(ANDROID_VULKAN_LIBRARY vulkan REQUIRED)
set(Vulkan_LIBRARY "${ANDROID_VULKAN_LIBRARY}" CACHE FILEPATH "" FORCE)
set(Vulkan_INCLUDE_DIR "${VENDOR}/vulkan-include" CACHE PATH "" FORCE)
# Latest CI fix is already applied. No user-side Python patch step is needed.
include("${CMAKE_CURRENT_LIST_DIR}/cmake/HostGlslc.cmake")
set(SPIRV_HEADERS_INCLUDE_DIR "${VENDOR}/SPIRV-Headers/include" CACHE PATH "" FORCE)
set(SPIRV_HEADERS_INCLUDE_DIRS "${VENDOR}/SPIRV-Headers/include" CACHE PATH "" FORCE)
# llama.cpp calls find_package(SPIRV-Headers CONFIG REQUIRED); supply the config from
# cmake/spirv-headers, which resolves to the vendored checkout that setup.sh clones.
set(SPIRV_HEADERS_ROOT "${VENDOR}/SPIRV-Headers" CACHE PATH "" FORCE)
set(SPIRV-Headers_DIR "${CMAKE_CURRENT_LIST_DIR}/cmake/spirv-headers" CACHE PATH "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_OPENCL ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_USE_ADRENO_BIN_KERNELS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
add_subdirectory(llama.cpp "${CMAKE_BINARY_DIR}/llama")
target_include_directories(ggml-vulkan PRIVATE "${VENDOR}/SPIRV-Headers/include")
target_link_libraries(ggml-opencl PRIVATE OpenCL::OpenCL)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/ggml/include")
check_cxx_source_compiles("#include <ggml-backend.h>\nint main() { auto t=GGML_BACKEND_DEVICE_TYPE_IGPU;return (int)t; }" LOCALLLM_HAS_IGPU)
unset(CMAKE_REQUIRED_INCLUDES)
add_library(llama_jni SHARED llama_jni.cpp)
target_compile_definitions(llama_jni PRIVATE LOCALLLM_HAS_IGPU=$<BOOL:${LOCALLLM_HAS_IGPU}>)
target_link_libraries(llama_jni PRIVATE llama ggml OpenCL::OpenCL android log dl)
target_link_options(llama_jni PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
